! =====================================================================
! R-EDGE (Cisco ISR 2911) - router de borde: inter-VLAN, ACL, NAT y VPN IPsec
! Packet Tracer 8.2.x. Pegar en modo "configure terminal".
! G0/0 = WAN hacia R-ISP | G0/1 = troncal 802.1Q hacia SW-CORE
! Requiere licencia securityk9:  license boot module c2900 technology-package securityk9
!                                (luego: write / reload)
! =====================================================================
hostname R-EDGE
no ip domain-lookup
ip domain-name act4.local
service password-encryption
security passwords min-length 10
enable algorithm-type scrypt secret Act4-Enable-2026
username admin privilege 15 algorithm-type scrypt secret Act4-Admin-2026
username vpnuser1 secret Act4-Vpn-User1
login block-for 120 attempts 3 within 60
banner motd ^ Acceso restringido. Actividad monitoreada. ^
!
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
!
! ---------------- WAN ----------------
interface GigabitEthernet0/0
 description WAN hacia ISP
 ip address 200.10.10.2 255.255.255.252
 ip nat outside
 ip access-group ACL-WAN-IN in
 no cdp enable
 no shutdown
ip route 0.0.0.0 0.0.0.0 200.10.10.1
!
! ---------------- Troncal y subinterfaces (router-on-a-stick) ----------------
interface GigabitEthernet0/1
 description Troncal 802.1Q hacia SW-CORE
 no ip address
 no shutdown
interface GigabitEthernet0/1.10
 description VLAN10-DMZ
 encapsulation dot1Q 10
 ip address 172.16.10.1 255.255.255.0
 ip nat inside
 ip access-group ACL-DMZ-IN in
interface GigabitEthernet0/1.20
 description VLAN20-INTERNA
 encapsulation dot1Q 20
 ip address 172.16.20.1 255.255.255.0
 ip nat inside
 ip access-group ACL-INTERNA-IN in
interface GigabitEthernet0/1.30
 description VLAN30-IOT
 encapsulation dot1Q 30
 ip address 172.16.30.1 255.255.255.0
 ip nat inside
 ip access-group ACL-IOT-IN in
interface GigabitEthernet0/1.99
 description VLAN99-GESTION
 encapsulation dot1Q 99 native
 ip address 172.16.99.1 255.255.255.0
 ip access-group ACL-GESTION-IN in
!
! ---------------- DHCP (INTERNA e IoT; la DMZ usa IP fija) ----------------
ip dhcp excluded-address 172.16.20.1 172.16.20.20
ip dhcp excluded-address 172.16.30.1 172.16.30.20
ip dhcp pool INTERNA
 network 172.16.20.0 255.255.255.0
 default-router 172.16.20.1
 dns-server 172.16.10.11
ip dhcp pool IOT
 network 172.16.30.0 255.255.255.0
 default-router 172.16.30.1
 dns-server 172.16.10.11
!
! ---------------- Objetos de referencia ----------------
! 172.16.10.10 = SRV-WEB (DMZ, publicado en 200.10.10.2:443)
! 172.16.10.11 = SRV-DNS (DMZ)
! 172.16.10.20 = BROKER-IOT MQTT/TLS 8883 (DMZ)
! 172.16.20.10 = SRV-ARCHIVOS (INTERNA)
! 172.16.20.50 = PC-ADMIN (INTERNA, unico equipo que administra)
! 10.10.50.0/24 = pool de clientes VPN
!
! ---------------- ACL de minimo privilegio (se aplican de entrada en cada VLAN) ----
! DMZ: nunca inicia conexiones hacia INTERNA, IoT ni GESTION. Solo responde y sale a
! Internet por actualizaciones (80/443) y DNS.
ip access-list extended ACL-DMZ-IN
 remark Respuestas a sesiones iniciadas desde INTERNA/VPN
 permit tcp 172.16.10.0 0.0.0.255 172.16.20.0 0.0.0.255 established
 permit tcp 172.16.10.0 0.0.0.255 10.10.50.0 0.0.0.255 established
 permit tcp host 172.16.10.20 172.16.30.0 0.0.0.255 established
 permit udp host 172.16.10.11 eq domain 172.16.20.0 0.0.0.255
 permit udp host 172.16.10.11 eq domain 172.16.30.0 0.0.0.255
 remark Bloqueo explicito hacia redes internas
 deny   ip 172.16.10.0 0.0.0.255 172.16.20.0 0.0.0.255 log
 deny   ip 172.16.10.0 0.0.0.255 172.16.30.0 0.0.0.255 log
 deny   ip 172.16.10.0 0.0.0.255 172.16.99.0 0.0.0.255 log
 remark Salida a Internet minima
 permit tcp 172.16.10.0 0.0.0.255 any eq 80
 permit tcp 172.16.10.0 0.0.0.255 any eq 443
 permit tcp host 172.16.10.10 eq 443 any established
 permit udp host 172.16.10.11 any eq domain
 deny   ip any any log
!
! INTERNA: usa servicios publicados en DMZ, sale a Internet; NO entra a IoT
! (salvo PC-ADMIN por HTTPS para administrar camaras) ni a GESTION (salvo PC-ADMIN).
ip access-list extended ACL-INTERNA-IN
 permit udp any eq bootpc any eq bootps
 permit tcp host 172.16.20.50 host 172.16.20.1 eq 22
 permit tcp 172.16.20.0 0.0.0.255 host 172.16.10.10 eq 443
 permit udp 172.16.20.0 0.0.0.255 host 172.16.10.11 eq domain
 permit tcp host 172.16.20.50 172.16.10.0 0.0.0.255 eq 22
 permit tcp host 172.16.20.50 172.16.30.0 0.0.0.255 eq 443
 permit tcp host 172.16.20.50 172.16.99.0 0.0.0.255 eq 22
 permit tcp 172.16.20.0 0.0.0.255 10.10.50.0 0.0.0.255 established
 deny   ip 172.16.20.0 0.0.0.255 172.16.10.0 0.0.0.255 log
 deny   ip 172.16.20.0 0.0.0.255 172.16.30.0 0.0.0.255 log
 deny   ip 172.16.20.0 0.0.0.255 172.16.99.0 0.0.0.255 log
 permit tcp 172.16.20.0 0.0.0.255 any eq 80
 permit tcp 172.16.20.0 0.0.0.255 any eq 443
 permit udp 172.16.20.0 0.0.0.255 any eq domain
 permit icmp 172.16.20.0 0.0.0.255 any echo
 deny   ip any any log
!
! IoT: solo habla con su broker MQTT/TLS y el DNS de la DMZ. Sin Internet directo,
! sin acceso a INTERNA (contiene un dispositivo comprometido: no hay movimiento lateral).
ip access-list extended ACL-IOT-IN
 permit udp any eq bootpc any eq bootps
 permit tcp 172.16.30.0 0.0.0.255 host 172.16.10.20 eq 8883
 permit udp 172.16.30.0 0.0.0.255 host 172.16.10.11 eq domain
 permit tcp 172.16.30.0 0.0.0.255 eq 443 host 172.16.20.50 established
 deny   ip 172.16.30.0 0.0.0.255 any log
!
! GESTION: solo responde al PC-ADMIN.
ip access-list extended ACL-GESTION-IN
 permit tcp 172.16.99.0 0.0.0.255 eq 22 host 172.16.20.50 established
 deny   ip any any log
!
! Acceso SSH a R-EDGE solo desde PC-ADMIN y la VPN
ip access-list standard ACL-VTY
 permit host 172.16.20.50
 permit 10.10.50.0 0.0.0.255
 deny   any log
line con 0
 login local
 exec-timeout 5 0
line vty 0 4
 login local
 transport input ssh
 access-class ACL-VTY in
 exec-timeout 5 0
!
! ---------------- WAN: solo HTTPS publicado e IPsec ----------------
ip access-list extended ACL-WAN-IN
 permit udp any host 200.10.10.2 eq isakmp
 permit udp any host 200.10.10.2 eq non500-isakmp
 permit esp any host 200.10.10.2
 permit tcp any host 200.10.10.2 eq 443
 permit tcp any any established
 permit udp any eq domain any
 permit icmp any any echo-reply
 deny   ip any any log
!
! ---------------- NAT ----------------
! PAT para las redes internas (excepto el trafico que va por la VPN)
ip access-list extended ACL-NAT
 deny   ip 172.16.0.0 0.0.255.255 10.10.50.0 0.0.0.255
 permit ip 172.16.10.0 0.0.0.255 any
 permit ip 172.16.20.0 0.0.0.255 any
ip nat inside source list ACL-NAT interface GigabitEthernet0/0 overload
! Publicacion del servidor web de la DMZ (solo 443)
ip nat inside source static tcp 172.16.10.10 443 200.10.10.2 443
!
! ---------------- VPN IPsec de acceso remoto (Easy VPN Server, IKEv1) ----------------
! Cliente: PC-REMOTO > Desktop > VPN  (GroupName ACT4-REMOTO, Key, usuario vpnuser1)
aaa new-model
aaa authentication login default local
aaa authentication login VPN-AUTH local
aaa authorization network VPN-GRUPO local
!
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 5
 lifetime 28800
!
ip local pool POOL-VPN 10.10.50.10 10.10.50.100
! Split tunnel: el cliente solo enruta por el tunel las redes corporativas permitidas
ip access-list extended ACL-SPLIT
 permit ip 172.16.10.0 0.0.0.255 any
 permit ip 172.16.20.0 0.0.0.255 any
!
crypto isakmp client configuration group ACT4-REMOTO
 key Act4-Grupo-PSK-2026
 pool POOL-VPN
 acl ACL-SPLIT
 dns 172.16.10.11
!
crypto ipsec transform-set TS-AES256 esp-aes 256 esp-sha-hmac
!
crypto dynamic-map DYN-VPN 10
 set transform-set TS-AES256
 reverse-route
!
crypto map MAPA-VPN client authentication list VPN-AUTH
crypto map MAPA-VPN isakmp authorization list VPN-GRUPO
crypto map MAPA-VPN client configuration address respond
crypto map MAPA-VPN 10 ipsec-isakmp dynamic DYN-VPN
!
interface GigabitEthernet0/0
 crypto map MAPA-VPN
!
! Usuarios VPN: llegan a INTERNA (archivos) y al web de DMZ; nunca a IoT/GESTION.
! (En IOS el trafico descifrado se filtra con la ACL de salida de cada subinterfaz)
ip access-list extended ACL-HACIA-IOT-OUT
 deny   ip 10.10.50.0 0.0.0.255 any log
 permit ip any any
interface GigabitEthernet0/1.30
 ip access-group ACL-HACIA-IOT-OUT out
!
end
write memory
